02How to check a factorial of a number using java with html?

CODE

<html>
<head>
<script>
function factorial()
{
  let a=Number(document.getElementById("num").value);
  f=1;
  for(a;a>1;a--)
  {
    f=f*a;
  }
  alert(f);
}
</script>
</head>
<body>
ENTER THE NUMBER: <input type="text" id="num" name="num" value="6"><br>
<input type="button" value="click to check" onclick="factorial()">
</body>
</html>

11How to check whether person is eligible to vote or not using java script by using html?

CODE

<html>
<head>
<script>
function verify()
{
  let a=Number(document.getElementById("age").value);
  if(a>=18)
  {
    alert("eligible to vote");
  }
  else
  {
    alert("ineligible to vote");
  }
}
</script>
</head>
<body>
ENTER YOUR AGE: <input type="text" id="age" name="age" value="21">
<input type="button" value="click to check" onclick="verify()">
</body>
</html>
